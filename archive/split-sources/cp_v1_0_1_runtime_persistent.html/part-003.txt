    const SignalClassifier = {
        classify(signal, ctx) {
            const r = { clarity: 'clear', intent_presence: 'no_action', context_fit: 'in_scope', agent_request: false };
            const l = signal.toLowerCase().trim();
            if (l.length < 3) { r.clarity = 'ambiguous'; return r; }
            if (/^(ask agent|get advice|what do you think|help me with|any suggestions)/i.test(l)) { r.agent_request = true; r.intent_presence = 'action_requested'; return r; }
            if (/^(maybe|perhaps|i guess|not sure|hmm)/i.test(l)) { r.clarity = 'ambiguous'; return r; }
            if (/^(i want|i need|please|create|make|generate|lock in|commit|mark)/i.test(l)) r.intent_presence = 'action_requested';
            /* FIX 1 (DPT-17): Added "off topic" and "different subject" to scope boundary patterns */
            if (ctx && /remind me to|by the way|unrelated|off topic|different subject/i.test(l)) r.context_fit = 'out_of_scope';
            return r;
        }
    };

    const CPGuardian = {
        evaluatePostClassification(c) {
            if (c.clarity === 'ambiguous') return { decision: 'Clarify', reason: 'Ambiguous signal', friction_trigger: { trigger_type: 'ambiguous_signal', trigger_description: 'Signal needs clarification' } };
            if (c.context_fit === 'out_of_scope') return { decision: 'Clarify', reason: 'Out of scope', friction_trigger: { trigger_type: 'scope_boundary', trigger_description: 'Outside context' } };
            return { decision: 'Allow', reason: c.intent_presence === 'action_requested' ? 'Action requested' : 'No action needed', friction_trigger: null };
        },
        evaluatePreExecution(req) {
            const int = IntentStore.get(req.intent_id);
            if (!int) return { decision: 'Refuse', reason: 'PE-1: No intent', friction_trigger: { trigger_type: 'unverified_intent', trigger_description: 'No intent' } };
            if (!['verified', 'active'].includes(int.state)) return { decision: 'Refuse', reason: 'PE-2: Intent not verified', friction_trigger: { trigger_type: 'unverified_intent', trigger_description: 'Intent state: ' + int.state } };
            const ctx = ContextStore.get(req.context_id);
            if (!ctx || ctx.state !== 'open') return { decision: 'Refuse', reason: 'PE-4: Context not open', friction_trigger: { trigger_type: 'scope_boundary', trigger_description: 'Context closed' } };
            if (req.agent_id) {
                if (!ctx.allowed_agents.includes(req.agent_id)) return { decision: 'Refuse', reason: 'PE-5: Agent not allowed', friction_trigger: { trigger_type: 'permission_violation', trigger_description: 'Agent not in allowed list' } };
                const agent = AgentStore.get(req.agent_id);
                if (!agent || agent.state !== 'active') return { decision: 'Refuse', reason: 'PE-6: Agent not active', friction_trigger: { trigger_type: 'permission_violation', trigger_description: 'Agent inactive' } };
                if (agent.agent_type === 'advisory' && ['modify', 'external'].includes(req.action_type)) return { decision: 'Refuse', reason: 'PE-7: Advisory cannot ' + req.action_type, friction_trigger: { trigger_type: 'permission_violation', trigger_description: 'Advisory agent restricted' } };
            }
            if (FrictionStore.getActive().length > 0) return { decision: 'Refuse', reason: 'PE-10: Unresolved friction', friction_trigger: { trigger_type: 'conflict_detected', trigger_description: 'Resolve friction first' } };
            return { decision: 'Allow', reason: 'PE-11: All checks passed', friction_trigger: null };
        }
    };

    const AdvisoryAgent = {
        async generateResponse(agentId, input, ctx, intent) {
            await new Promise(r => setTimeout(r, 500 + Math.random() * 500));
            const nodes = NodeStore.getNodesByContext(ctx.context_id);
            let resp = `Based on "${ctx.description}" and intent "${intent.declaration_text}", `;
            if (/next|proceed|continue/i.test(input)) {
                resp += nodes.length === 0 ? `consider committing your first node—a foundational fact or decision.` : `you have ${nodes.length} node(s). Consider adding supporting elements or reviewing pivots.`;
            } else if (/structure|organize/i.test(input)) {
                resp += `try: 1) Core concepts as fact nodes, 2) Key decisions as decision nodes, 3) Boundaries as constraint nodes.`;
            } else if (/help|advice|suggest/i.test(input)) {
                resp += `I can suggest structures, identify themes, recommend nodes vs pivots. I cannot create nodes—you must commit them explicitly.`;
            } else {
                resp += `noted: "${input}". ` + (intent.intent_type === 'explore' ? 'Gather info before committing. Use pivots for items to revisit.' : 'How can I help?');
            }
            return resp + '\n\n(Advisory only—commit nodes explicitly.)';
        }
