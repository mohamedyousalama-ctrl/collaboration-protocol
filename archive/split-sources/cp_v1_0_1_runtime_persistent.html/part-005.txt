            const p = Session.pendingVerification;
            if (p.type === 'context_and_intent') {
                const ctx = createContext(p.data.description); ContextStore.create(ctx); Session.activeContextId = ctx.context_id;

            } else if (p.type === 'intent') {

                const node = createNode(Session.activeContextId, p.data.content, p.data.type); NodeStore.createNode(node);
                LogStore.append({ event_type: 'node', event_subtype: 'created', object_id: node.node_id, context_id: Session.activeContextId, intent_id: Session.activeIntentId, agent_id: null, user_id: 'user_01', details: 'Node: ' + node.content, reason: null });

                UI.updateNodes();
            }
            FrictionStore.getActive().forEach(f => { f.state = 'resolved'; f.resolution = 'user_confirmed'; f.resolved_at = timestamp(); FrictionStore.update(f.friction_id, f); LogStore.append({ event_type: 'friction', event_subtype: 'resolved', object_id: f.friction_id, context_id: Session.activeContextId, intent_id: Session.activeIntentId, agent_id: null, user_id: 'user_01', details: 'Friction resolved', reason: null }); });
            Session.pendingVerification = null; Session.systemState = 'idle'; Session.persist(); UI.updateStatus();
        },
        handleClarification(text) {
