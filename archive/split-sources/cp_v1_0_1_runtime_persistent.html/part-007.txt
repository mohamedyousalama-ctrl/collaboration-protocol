        newContext() {
            const desc = prompt('Context description:'); if (!desc) return;
            const enableAgent = confirm('Enable advisory agent?\n\nOK = Yes, Cancel = No');
            /* FIX 3 (EDGE-17): Agent now suspended when context archived via New Context */
            if (Session.activeContextId) { 
                const old = ContextStore.get(Session.activeContextId); 
                if (old) { 
                    /* Suspend agent before archiving context */
                    if (Session.activeAgentId) {
                        const oldAgent = AgentStore.get(Session.activeAgentId);
                        if (oldAgent && oldAgent.state === 'active') {
                            oldAgent.state = 'suspended';
                            AgentStore.update(oldAgent.agent_id, oldAgent);
                            LogStore.append({ event_type: 'agent', event_subtype: 'state_changed', object_id: oldAgent.agent_id, context_id: old.context_id, intent_id: Session.activeIntentId, agent_id: oldAgent.agent_id, user_id: 'user_01', details: 'Agent suspended', reason: 'Context archived for new context' });
                        }
                    }
                    old.state = 'archived'; 
                    ContextStore.update(old.context_id, old); 
                    LogStore.append({ event_type: 'context', event_subtype: 'state_changed', object_id: old.context_id, context_id: old.context_id, intent_id: null, agent_id: null, user_id: 'user_01', details: 'Context archived', reason: 'New context' }); 
                } 
            }
            const ctx = createContext(desc); ContextStore.create(ctx); Session.activeContextId = ctx.context_id; Session.activeIntentId = null; Session.activeAgentId = null; Session.systemState = 'idle'; Session.pendingVerification = null; Session.persist();
            LogStore.append({ event_type: 'context', event_subtype: 'created', object_id: ctx.context_id, context_id: ctx.context_id, intent_id: null, agent_id: null, user_id: 'user_01', details: 'Context: ' + ctx.description, reason: null });
            let msg = 'New context:\n- ' + ctx.context_id + '\n- ' + ctx.description + '\n';
            if (enableAgent) {
                const agent = createAgent(ctx.context_id, 'CP Advisory Agent'); AgentStore.create(agent); ctx.allowed_agents.push(agent.agent_id); ContextStore.update(ctx.context_id, ctx); Session.activeAgentId = agent.agent_id; Session.persist();
                LogStore.append({ event_type: 'agent', event_subtype: 'created', object_id: agent.agent_id, context_id: ctx.context_id, intent_id: null, agent_id: agent.agent_id, user_id: 'user_01', details: 'Agent: ' + agent.name, reason: null });
                LogStore.append({ event_type: 'agent', event_subtype: 'registered', object_id: agent.agent_id, context_id: ctx.context_id, intent_id: null, agent_id: agent.agent_id, user_id: 'user_01', details: 'Agent registered', reason: null });
                msg += '\n- Agent: ' + agent.name + ' ENABLED';
            }
            msg += '\n\nDeclare your intent.';
